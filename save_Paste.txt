local RealWalkSpeed, RealJumpPower = 16, 50
local HumanoidSpoof = setmetatable({}, {__mode = "k"})

if hookmetamethod and getrawmetatable and setreadonly and newcclosure then
    local mt = getrawmetatable(game)
    local oldNamecall = mt.__namecall
    local oldIndex = mt.__index
    local oldNewIndex = mt.__newindex
    setreadonly(mt, false)

    mt.__namecall = newcclosure(function(self, ...)
        local method = getnamecallmethod()
        local args = {...}
        if method == "FindFirstChild" or method == "FindFirstChildOfClass" then
            if tostring(self) == "CoreGui" or self == CoreGui then
                local n = tostring(args[1])
                if n == "RakanHub" or n == "RakanToggle" or n == "RakanMain" then
                    return nil
                end
            end
        end
        if method == "GetDescendants" then
            local result = oldNamecall(self, ...)
            if type(result) == "table" then
                local filtered = {}
                for _, v in ipairs(result) do
                    local n = (typeof and typeof(v) == "Instance") and v.Name or tostring(v)
                    if n ~= "RakanHub" and n ~= "RakanToggle" and n ~= "RakanMain" then
                        table.insert(filtered, v)
                    end
                end
                return filtered
            end
        end
        return oldNamecall(self, ...)
    end)

    mt.__index = newcclosure(function(self, key)
        if typeof(self) == "Instance" and self:IsA("Humanoid") then
            if key == "WalkSpeed" then return RealWalkSpeed end
            if key == "JumpPower" then return RealJumpPower end
        end
        return oldIndex(self, key)
    end)

    mt.__newindex = newcclosure(function(self, key, value)
        if typeof(self) == "Instance" and self:IsA("Humanoid") then
            if key == "WalkSpeed" or key == "JumpPower" then
                local stored = HumanoidSpoof[self] or {}
                stored[key] = value
                HumanoidSpoof[self] = stored
                return oldNewIndex(self, key, (key == "WalkSpeed" and 16 or 50))
            end
        end
        return oldNewIndex(self, key, value)
    end)

    setreadonly(mt, true)
end

local function setRealSpeed(h, v)
    if not h then return end
    local mt = getrawmetatable(game)
    local oldNewIndex = mt.__newindex
    setreadonly(mt, false)
    oldNewIndex(h, "WalkSpeed", v)
    setreadonly(mt, true)
    HumanoidSpoof[h] = HumanoidSpoof[h] or {}
    HumanoidSpoof[h].WalkSpeed = v
end

local function setRealJump(h, v)
    if not h then return end
    local mt = getrawmetatable(game)
    local oldNewIndex = mt.__newindex
    setreadonly(mt, false)
    oldNewIndex(h, "JumpPower", v)
    setreadonly(mt, true)
    HumanoidSpoof[h] = HumanoidSpoof[h] or {}
    HumanoidSpoof[h].JumpPower = v
end

        tw(dot, {Position = UDim2.new(v and 1 or 0, v and -19 or 3, 0.5, -8),

            knob.BackgroundColor3 = Theme.Accent
            dot.Position = UDim2.new(1, -19, 0.5, -8)
            dot.BackgroundColor3 = Theme.AccentGlow
        end)
    end
    return {Set = set, Get = function() return state end}

local function makeSlider(parent, txt, minV, maxV, default, cb)
    local f = Instance.new("Frame")
    f.Size = UDim2.new(1, 0, 0, 56)
    f.BackgroundColor3 = Theme.BgInput
    f.Parent = parent
    corner(10, f)
    stroke(Theme.Border, 1, f)

    local lbl = Instance.new("TextLabel")
    lbl.Size = UDim2.new(1, -100, 0, 20)
    lbl.Position = UDim2.new(0, 12, 0, 6)
    lbl.BackgroundTransparency = 1
    lbl.Text = txt
    lbl.TextColor3 = Theme.Text
    lbl.Font = Enum.Font.GothamMedium
    lbl.TextSize = 13
    lbl.TextXAlignment = Enum.TextXAlignment.Left
    lbl.Parent = f

    local val = Instance.new("TextLabel")
    val.Size = UDim2.new(0, 60, 0, 20)
    val.Position = UDim2.new(1, -72, 0, 6)
    val.BackgroundTransparency = 1
    val.Text = tostring(default or minV)
    val.TextColor3 = Theme.AccentGlow
    val.Font = Enum.Font.GothamBold
    val.TextSize = 13
    val.TextXAlignment = Enum.TextXAlignment.Right
    val.Parent = f

    local bar = Instance.new("Frame")
    bar.Size = UDim2.new(1, -24, 0, 6)
    bar.Position = UDim2.new(0, 12, 0, 38)
    bar.BackgroundColor3 = Theme.Border
    bar.Parent = f
    corner(3, bar)

    local fill = Instance.new("Frame")
    fill.Size = UDim2.new(0, 0, 1, 0)
    fill.BackgroundColor3 = Theme.Accent
    fill.Parent = bar
    corner(3, fill)

    local click = Instance.new("TextButton")
    click.Size = UDim2.new(1, 24, 0, 20)
    click.Position = UDim2.new(0, -12, 0, 28)
    click.BackgroundTransparency = 1
    click.Text = ""
    click.Parent = f

    local current = default or minV
    local function update(x)
        local rel = math.clamp((x - bar.AbsolutePosition.X) / bar.AbsoluteSize.X, 0, 1)
        current = math.floor((minV + (maxV - minV) * rel) + 0.5)
        val.Text = tostring(current)
        tw(fill, {Size = UDim2.new(rel, 0, 1, 0)}, 0.05)
        if cb then cb(current) end
    end

    click.MouseButton1Down:Connect(function()
        update(UserInputService:GetMouseLocation().X)
        local moveConn
        moveConn = UserInputService.InputChanged:Connect(function(inp)
            if inp.UserInputType == Enum.UserInputType.MouseMovement or inp.UserInputType == Enum.UserInputType.Touch then
                update(inp.Position.X)
            end
        end)
        local endConn
        endConn = UserInputService.InputEnded:Connect(function(inp)
            if inp.UserInputType == Enum.UserInputType.MouseButton1 or inp.UserInputType == Enum.UserInputType.Touch then
                if moveConn then moveConn:Disconnect() end
                if endConn then endConn:Disconnect() end
            end
        end)
    end)

    task.defer(function()
        local rel = (current - minV) / (maxV - minV)
        fill.Size = UDim2.new(rel, 0, 1, 0)
    end)
    return {Set = function(v) current = v; val.Text = tostring(v) end}
end

    tabs[name] = tab

    btn.MouseButton1Click:Connect(function()

            t.Page.Visible = (n == name)

        end
        page.Visible = true
        tw(btn, {BackgroundColor3 = Theme.Accent, TextColor3 = Theme.Text}, 0.15)

    end)

            if p ~= LocalPlayer and p.Character then
                local part = p.Character:FindFirstChild(Aim.Part)
                local hum = p.Character:FindFirstChildOfClass("Humanoid")
                if part and hum and hum.Health > 0 then
                    if not (Aim.TeamCheck and p.Team == LocalPlayer.Team) then
                        local sp, onScreen = Workspace.CurrentCamera:WorldToViewportPoint(part.Position)
                        if onScreen then
                            local d = (Vector2.new(sp.X, sp.Y) - mouse).Magnitude
                            if d < best then best = d; closest = p end
                        end
                    end
                end
            end
        end
        return closest
    end

    RunService.RenderStepped:Connect(function()
        if FOVCircle then
            FOVCircle.Visible = Aim.ShowFOV
            FOVCircle.Position = UserInputService:GetMouseLocation()
            FOVCircle.Radius = Aim.FOV
        end
        if Aim.Enabled and UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then
            local target = getClosest()
            if target and target.Character then
                local part = target.Character:FindFirstChild(Aim.Part)
                if part then
                    local cam = Workspace.CurrentCamera
                    local desired = CFrame.new(cam.CFrame.Position, part.Position)
                    cam.CFrame = cam.CFrame:Lerp(desired, Aim.Smooth)
                end
            end
        end
    end)
end

-- DRAG SUPPORT
-- ==========================================================
local function makeDraggable(obj, handle)
    local dragging, dragStart, startPos
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = obj.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            obj.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end
